 def test_original_result_logic_is_retained(self):
  """Remove only declared additions, then compare complete old validator AST."""
  old=ast.parse((R/'minicpm5_original32_layer0_full_bf16_cpu_runner_v1.py').read_text())
  new=ast.parse((Q/'minicpm5_layer1_runtime_results_v1.py').read_text())
  oldfn=next(n for n in old.body if getattr(n,'name',None)=='validate_rank_result')
  newfn=next(n for n in new.body if getattr(n,'name',None)=='validate_rank_result')
  class Normalize(ast.NodeTransformer):
   def visit_Constant(self,node):
    if type(node.value)is str:
     node.value=node.value.replace('minicpm5_layer1_observation_cpu_','minicpm5_original32_layer0_full_bf16_cpu_')
    return node
   def visit_Expr(self,node):
    if isinstance(node.value,ast.Call):
     text=ast.unparse(node)
     if 'complete84_state_snapshots_written'in text or text.startswith('observation_metadata('):return None
     if 'full_cache_payloads_written'in text:
      node.value.args[1].elts[0]=ast.Constant(0)
    return self.generic_visit(node)
   def visit_Assign(self,node):
    if any(isinstance(t,ast.Name)and t.id in ('previous_observation','original_ref','original_call')for t in node.targets):
     return None
    return self.generic_visit(node)
  self.assertEqual(ast.dump(oldfn),ast.dump(Normalize().visit(newfn)))
