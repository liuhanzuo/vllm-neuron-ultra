def classify(exited,terminal,failure,post_review):
 code=exited['actual_exit']['exit_code'];owner=exited['owner'];request=exited['request']
 if terminal is not None:
  require(terminal['schema']=='dory_rotary_actual_terminal_v1'and terminal['complete']is True,'Complete retained arithmetic terminal')
  same(terminal['owner'],owner,'Native terminal owner');same(terminal['request'],request,'Native terminal request')
  require(terminal['actual_self_exit_claimed']is False and terminal['native_reference_qualified']is False and terminal['neuron_baseline_qualified']is False and terminal['new_speedup_claim']is False,'Local arithmetic never model qualification')
  require(post_review is not None and post_review['complete']is True,'Complete independent post-exit review')
  same(post_review['result'],terminal['result'],'Same original saved result')
  same(terminal['arithmetic_passed'],post_review['observer_invariant']and post_review['arithmetic_parity'],'Original observer and arithmetic outcome')
  same(terminal['reference_replay_passed'],post_review['all_reference_outputs_reproduced'],'Original reference replay outcome')
  same(terminal['candidate_replay_passed'],post_review['all_candidate_outputs_reproduced'],'Original candidate replay outcome')
  require(all(type(terminal[k])is bool for k in ('arithmetic_passed','reference_replay_passed','candidate_replay_passed','passed')),'Actual terminal booleans')
  same(terminal['passed'],all(terminal[k]for k in ('arithmetic_passed','reference_replay_passed','candidate_replay_passed')),'Derived local pass classification')
 if failure is not None:
  same(failure['owner'],owner,'Failure original owner');same(failure['request'],request,'Failure original request')
  require(failure['actual_self_exit_claimed']is False and failure['native_reference_qualified']is False and failure['new_speedup_claim']is False,'Failure not promoted')
  require(type(failure['error_type'])is str and type(failure['error'])is str,'Original failure retained')
 if code==0:require(terminal is not None and terminal['passed']is True and failure is None,'Zero exit needs complete successful local result and no failure')
 else:require(failure is not None,'Nonzero execution must retain original failure')
 return dict(actual_exit_code=code,arithmetic_qualified=bool(code==0 and terminal is not None and terminal['passed']),complete_saved_data_reviewed=post_review is not None,native_reference_qualified=False,neuron_baseline_qualified=False,producer_qualified=False,new_speedup_claim=False)

def text(path,expected=None):

def source_paths():
 paths=set(map(Path,runner.source_paths()))|{SELF,Q/'check_dory_rotary_completed_history_v1.py'};todo=list(paths);seen=set()
 while todo:
  path=todo.pop()
  if path in seen:continue
  seen.add(path)
  if path.suffix!='.py':continue
  for node in ast.walk(ast.parse(path.read_text())):
   names=[x.name for x in node.names]if isinstance(node,ast.Import)else[node.module]if isinstance(node,ast.ImportFrom)and node.module and not node.level else[]
   for name in names:
    for base in (R,Q):
     child=base/(name.replace('.','/')+'.py')
     if child.is_file()and child not in paths:paths.add(child);todo.append(child)
 return sorted(map(str,paths))


def write_once(path,value):
 raw=(json.dumps(value,indent=2,allow_nan=False)+'\n').encode();require(len(raw)<32*2**20,'Bounded Dory finalizer file')
 fd=os.open(path,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600)
 with os.fdopen(fd,'wb')as f:f.write(raw);f.flush();os.fsync(f.fileno())
 p.metadata.sync_parent(path);return dict(path=str(path),sha256=hashlib.sha256(raw).hexdigest())


def finalize(exit_ref,finalizer_review_ref):

 same(approval['actual_exit'],exit_ref,'Review binds genuine external exit');same(approval['request'],request_ref,'Finalizer exact request');same(approval['budgets'],BUDGET,'Original finalizer resource limits')
 same(set(approval['source_sha256']),set(source_paths()),'Complete reviewed finalizer source closure')
 for path,sha in approval['source_sha256'].items():same(ref(path)['sha256'],sha,'Exact reviewed finalizer source')
 audit_file=audit_path(request['run_id']);catalog_file=catalog_path(request['run_id']);post_file=Q/('dory-rotary-'+request['run_id']+'-post-exit-raw-review-v1.json')
 require(not any(x.exists()for x in (audit_file,catalog_file,post_file)),'Never overwrite finalizer evidence')

 try:
  original=runner.validate_terminal(request['catalog'],request['finalizer_exit']);prior.physical_drain(original);p.low.gone(exited['owner']);p.low.active_heavy()
  os.sched_setaffinity(0,set(BUDGET['cpus']));os.nice(19);resource.setrlimit(resource.RLIMIT_AS,(BUDGET['address_space_bytes'],)*2);resource.setrlimit(resource.RLIMIT_CPU,(BUDGET['cpu_seconds'],BUDGET['cpu_seconds']+5));resource.setrlimit(resource.RLIMIT_FSIZE,(BUDGET['file_bytes'],)*2)
  def alarm(signum,frame):raise TimeoutError('Dory post-exit finalizer wall budget')
  old_alarm=signal.signal(signal.SIGALRM,alarm);signal.alarm(BUDGET['wall_seconds'])
  def guard():
   require(time.monotonic()-started<BUDGET['wall_seconds'],'Bounded post-exit work');p.low.lock_proof(owner,lock);runner.ensure_observer();runner.usage(root)
   fs=os.statvfs(R);require(fs.f_bavail*fs.f_frsize>=BUDGET['root_floor_bytes'],'Original root free-space floor')
  guard();metadata_refs={str(path.relative_to(root)):ref(path)for path in sorted(root.rglob('*.json'))}
  terminal=load_small(metadata_refs['terminal.json'])if 'terminal.json'in metadata_refs else None;failure=load_small(metadata_refs['failure.json'])if 'failure.json'in metadata_refs else None;post=None;post_ref=None
  if terminal is not None:
   same(terminal['result']['path'],str(root/'comparison/result.json'),'Actual original full result')
   from review_dory_rotary_saved_boundaries_v1 import review as independent
   post=independent(terminal['result'],guard);same(post,load_small(terminal['independent_review']),'Exact independent review after native exit')
   post_ref=write_once(post_file,post)
  outcome=classify(exited,terminal,failure,post);guard();p.low.gone(exited['owner']);same(p.canonical(),request['canonical_sha256'],'Canonical state unchanged')
  audit=dict(schema='dory_rotary_root_terminal_v1',reviewer='/root',observed_at=datetime.datetime.now(datetime.timezone.utc).isoformat(),actual_exit=exit_ref,request=request_ref,source_review=review_ref,previous_catalog=request['catalog'],finalizer_source_review=finalizer_review_ref,finalizer_owner=owner,finalizer_self_exit_claimed=False,next_admission_must_verify_finalizer_exit=True,physical_drain_passed=True,metadata=metadata_refs,post_exit_raw_review=post_ref,outcome=outcome,original_science=request['science'],all_15_model_families_remain_in_scope=True,new_speedup_claim=False)
  audit_ref=write_once(audit_file,audit);pins=dict(request['source_sha256'])
  for path,sha in approval['source_sha256'].items():require(path not in pins or pins[path]==sha,'No conflicting finalizer source');pins[path]=sha
  for item in [request['catalog'],exit_ref,request_ref,review_ref,finalizer_review_ref,audit_ref,*metadata_refs.values(),*([post_ref]if post_ref else[])]:require(item['path']not in pins or pins[item['path']]==item['sha256'],'No conflicting evidence pin');pins[item['path']]=item['sha256']
  value=compose(original,request['catalog'],audit_ref,exit_ref,request_ref,review_ref,finalizer_review_ref,outcome,exited['owner'],owner,pins);validate(value);catalog_ref=write_once(catalog_file,value);guard()
  print(json.dumps(dict(catalog=catalog_ref,audit=audit_ref,finalizer_owner=owner,outcome=outcome,actual_self_exit_claimed=False)),flush=True)
